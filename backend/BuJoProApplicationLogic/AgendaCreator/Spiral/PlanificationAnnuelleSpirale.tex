\newcommand{\monthCalendar}[3]{
    \StrBefore{#1}{,}[\xCoord] % Extract x coordinate
    \StrBehind{#1}{,}[\yCoord]
    \node [text width=10cm, align=center, fill=white, minimum height=3.5cm] at (\xCoord + 2.3 , \yCoord - 0.5) {
    };
    \node [font=\fontsize{8pt}\selectfont, text width=4.75cm, align=center] at (\xCoord + 2.2, \yCoord + 1) {
        \texttt{#3}
    };
    \pgfmathsetmacro{\yposition}{\yCoord + 0.5}
    \pgfmathsetmacro{\xposition}{\xCoord}
    \foreach \day [count=\j] in #2 {
        \ifthenelse{\equal{\day}{XX}}{}{
            \node at (\xposition, \yposition) {\texttt{\day}};
        }
        \pgfmathparse{int(mod(\j,7)==0)}
        \ifnum\pgfmathresult=1
            \pgfmathparse{\yposition - 0.5} % Perform arithmetic
            \global\let\yposition\pgfmathresult 
            \pgfmathparse{\xCoord + 0 } % Perform arithmetic
            \global\let\xposition\pgfmathresult
        \else 
            \pgfmathparse{\xposition + 0.75}
            \global\let\xposition\pgfmathresult
        \fi
    }
}
